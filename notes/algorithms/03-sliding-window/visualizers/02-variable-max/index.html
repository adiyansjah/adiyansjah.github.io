Algorithm Pseudocode
function lengthOfLongestSubstring(s):
charSet = new Set()
left = 0
maxLen = 0
for right = 0 to s.length - 1:
while s[right] in charSet:
charSet.remove(s[left])
left++
charSet.add(s[right])
if right - left + 1 > maxLen:
maxLen = right - left + 1
return maxLen
