Algorithm Pseudocode
function minSubArrayLen(target, arr):
left = 0
currentSum = 0
minLen = Infinity
for right = 0 to arr.length - 1:
currentSum += arr[right]
while currentSum >= target:
if right - left + 1 < minLen:
minLen = right - left + 1
bestWindow = [left, right]
currentSum -= arr[left]
left++
return minLen == Infinity ? 0 : minLen
