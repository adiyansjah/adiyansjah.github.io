Algorithm Pseudocode
function maxSumK(arr, k):
n = arr.length
if n < k: return -1
windowSum = sum(arr[0:k])
maxSum = windowSum
for right = k to n-1:
windowSum += arr[right]
windowSum -= arr[right - k]
if windowSum > maxSum:
maxSum = windowSum
return maxSum
