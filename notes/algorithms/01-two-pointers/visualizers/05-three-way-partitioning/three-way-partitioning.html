Algorithm
def sort_colors(nums):
low = 0 # next position for 0
mid = 0 # current position
high = len(nums) - 1 # next position for 2
while mid <= high:
if nums[mid] == 0:
swap(nums[low], nums[mid])
low += 1
mid += 1
elif nums[mid] == 2:
swap(nums[mid], nums[high])
high -= 1
# Don't increment mid!
else: # nums[mid] == 1
mid += 1
