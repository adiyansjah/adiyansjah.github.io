Algorithm
def merge(nums1, m, nums2, n):
p1 = m - 1 # last in nums1
p2 = n - 1 # last in nums2
write = m + n - 1 # write position
while p2 >= 0:
if p1 >= 0 and nums1[p1] > nums2[p2]:
nums1[write] = nums1[p1]
p1 -= 1
else:
nums1[write] = nums2[p2]
p2 -= 1
write -= 1
# nums1 elements already in place!
