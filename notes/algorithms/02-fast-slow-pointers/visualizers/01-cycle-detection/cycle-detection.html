Algorithm Pseudocode
function detectCycle(head):
slow = head
fast = head
// Phase 1: Detect cycle
while fast != null and fast.next != null:
slow = slow.next
fast = fast.next.next
if slow == fast:
// Phase 2: Find entry
slow = head
while slow != fast:
slow = slow.next
fast = fast.next
return slow // cycle entry
return null // no cycle
