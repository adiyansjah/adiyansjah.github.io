Algorithm Pseudocode
function nthFromEnd(head, n):
slow = head
fast = head
// Move fast n steps ahead
for i = 0 to n-1:
if fast == null: return null
fast = fast.next
// Move both until fast reaches end
while fast != null:
slow = slow.next
fast = fast.next
return slow // nth from end
function isPalindrome(head):
// Stage 1: Find middle
slow = head, fast = head
while fast and fast.next:
slow = slow.next
fast = fast.next.next
// Stage 2: Reverse second half
secondHalf = reverse(slow)
// Stage 3: Compare halves
p1 = head, p2 = secondHalf
while p2 != null:
if p1.val != p2.val: return false
p1 = p1.next, p2 = p2.next
return true
